    public int loadLibs(String libVCOMExamplePath, String libVCOMExampleJNIWrapperPath){
        
        if(libsLoaded == true){
            return V100_Types.ERROR_LIBS_ALREADY_LOADED; // libs already loaded
        }
                
        try{
            System.load(libVCOMExamplePath);        
            System.load(libVCOMExampleJNIWrapperPath);
        } catch (UnsatisfiedLinkError error){
            return V100_Types.ERROR_INTERNAL;
        } catch (SecurityException error) {
            return V100_Types.ERROR_INTERNAL;
        } catch (NullPointerException error){
            return V100_Types.ERROR_INTERNAL;
        }  
        
        libsLoaded = true;
        return 0;
    }

    public int V100IDGetSystemMetrics(V100_DEVICE_TRANSPORT_INFO pDev, List<V100_Types._MX00_DB_METRICS>pDBMetrics, int[] nNumDBsFound){
        if(libsLoaded == true){
            int rc = V100_ID_Get_System_Metrics_Size(pDev, nNumDBsFound);
            if (rc != V100_Types.GEN_OK) {
                return rc;
            }
            
            V100_Types types = new V100_Types();
            V100_Types._MX00_DB_METRICS[] pMetrics = new V100_Types._MX00_DB_METRICS[nNumDBsFound[0]];
            for (int i = 0; i < nNumDBsFound[0]; i++) {
                pMetrics[i] = types.new _MX00_DB_METRICS();
            }
            rc = V100_ID_Get_System_Metrics_Values(pDev, pMetrics);
            
            pDBMetrics.clear();
            for (int i = 0; i < nNumDBsFound[0]; i++) {
                pDBMetrics.add(pMetrics[i]);
            }
                
            return rc;
        }
        else{
            return V100_Types.ERROR_LIBS_NOT_LOADED;
        }
    }
